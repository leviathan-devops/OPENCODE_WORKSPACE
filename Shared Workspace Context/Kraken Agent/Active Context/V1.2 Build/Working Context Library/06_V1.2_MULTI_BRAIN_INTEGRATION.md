const DOMAIN_OWNERSHIP = {

  'thinking-state':   ['kraken-reasoning', 'kraken-system'],

  'workflow-state':   ['kraken-system', 'kraken-execution'],

## Priority Messenger API

```typescript
interface BrainMessage {
  from: string;
  to: string;
  type: 'context-inject' | 'gate-failure' | 'checkpoint' | 'override' | 'sync';
  priority: 'critical' | 'high' | 'normal' | 'low';
  payload: Record<string, unknown>;
  requiresAck: boolean;
}

// Usage
const messenger = createBrainMessenger();

// Planning Brain → Execution Brain: context injection
messenger.send({
  from: 'kraken-planning',
  to: 'kraken-execution',
  type: 'context-inject',
  priority: 'high',
  payload: { thinking-state: { spec-context: [...] } },
  requiresAck: false,
});

// Execution Brain → Subagent: override
messenger.send({
  from: 'kraken-execution',
  to: 'kraken-subagent',
  type: 'override',
  priority: 'critical',
  payload: { taskId: 'xyz', action: 'abort' },
  requiresAck: true,
});

// System Brain → All: gate failure
messenger.send({
  from: 'kraken-system',
  to: '*',  // broadcast
  type: 'gate-failure',
  priority: 'critical',
  payload: { gateId: 'output-retrieval-gate', taskId: 'xyz' },
  requiresAck: true,
});
```

## Brain Initialization

```typescript
async function initializeKrakenBrains() {
  // 1. Create shared infrastructure
  const stateStore = createStateStore(DOMAIN_OWNERSHIP);
  const messenger = createBrainMessenger();
  const evidenceCollector = createEvidenceCollector();
  
  // 2. Initialize Planning Brain (loads T2 Master)
  const planningBrain = await createPlanningBrain({
    stateStore,
    messenger,
    t2Master: loadT2MasterContext(),  // Shark T2 + Kraken extensions
  });
  
  // 3. Initialize Execution Brain
  const executionBrain = await createExecutionBrain({
    stateStore,
    messenger,
    subagentManager: createSubagentManager({...}),
  });
  
  // 4. Initialize System Brain
  const systemBrain = await createSystemBrain({
    stateStore,
    messenger,
    evidenceCollector,
    firewall: createKrakenFirewall(),
  });
  
  // 5. Initialize Council
  const council = await createRoundtableCouncil({
    alpha: createAlphaBrain({...}),
    beta: createBetaBrain({...}),
    gamma: createGammaBrain({...}),
    messenger,
  });
  
  // 6. Wire brains together
  messenger.linkBrain('kraken-planning', planningBrain);
  messenger.linkBrain('kraken-execution', executionBrain);
  messenger.linkBrain('kraken-system', systemBrain);
  messenger.linkBrain('alpha', council.alpha);
  messenger.linkBrain('beta', council.beta);
  messenger.linkBrain('gamma', council.gamma);
  
  return { planningBrain, executionBrain, systemBrain, council };
}
```

## Subagent-Manager Integration

```typescript
interface SubagentIntegration {
  // Execution Brain has override authority
  canOverride: true,
  
  // Subagent is autonomous between overrides
  autonomyLevel: 'high',
  
  // Override triggers
  overrideTriggers: [
    'output-not-retrieved',
    'focus-collision',
    'premature-completion',
    'context-stale',
  ],
  
  // Output retrieval is mandatory
  outputRetrievalEnforced: true,
}

// How Execution Brain supervises
async function superviseSubagent(taskId: string) {
  const status = await get_cluster_status(taskId);
  
  if (status.success && !status.outputsRetrieved) {
    // BLOCKED by L2 - outputs not retrieved
    throw new Error(`[EXECUTION_OVERRIDE] Task ${taskId} claims success but outputs not retrieved`);
  }
  
  if (status.state === 'blocked') {
    // Subagent blocked - assess if override needed
    const assessment = await assessBlockage(status.blockReason);
    if (assessment.shouldOverride) {
      // Send override to subagent
      await sendOverride(taskId, assessment.correctiveAction);
    }
  }
}
```
