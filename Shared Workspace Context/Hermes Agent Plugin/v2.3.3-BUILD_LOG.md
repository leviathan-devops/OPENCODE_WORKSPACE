## BUG #1 — e004: lint Tool Crashes with ESLint Config Error

**File:** `src/tools/quality.ts`, lines 7-25
**Error:** `ESLint couldn't find an eslint.config.(js|mjs|cjs) file`
**Severity:** CRITICAL — blocks all lint exercises

### Root Cause
`lint` executes `npx eslint target` via `spawnSync`. ESLint 10 (version installed) requires an explicit config file and returns exit code 2 with the error "couldn't find an eslint.config" when none exists. The tool does not handle this failure mode — it passes the error through as `Lint errors found:\n${result.stdout || result.stderr}`.

### Fix Required

```typescript
// BEFORE (line 18-23):
const cmd = args.fix ? ["npx", "eslint", target, "--fix"] : ["npx", "eslint", target];
const result = spawnSync(cmd[0], cmd.slice(1), { cwd: context.directory, encoding: "utf-8", timeout: 60000 });
if (result.status !== 0) {
  return `Lint errors found:\n${result.stdout || result.stderr || "Lint failed"}`;
}
return `✓ No lint errors found.`;

// AFTER:
const eslintArgs = args.fix ? ["--fix"] : [];
// First try with project eslint config
let result = spawnSync("npx", ["eslint", target, ...eslintArgs], { cwd: context.directory, encoding: "utf-8", timeout: 60000 });

// If ESLint fails with config error, create a minimal inline config and retry
if (result.status !== 0) {
  const stderr = result.stderr || "";
  if (stderr.includes("couldn't find an eslint.config") || stderr.includes("Cannot find module")) {
    // Write minimal config to tmp and use --config flag
    const tmpConfig = `/tmp/.eslintrc-minimal-${process.pid}.js`;
    const fs = require("fs");
    fs.writeFileSync(tmpConfig, "module.exports = { rules: { 'no-unused-vars': 'off', 'no-console': 'off' } };");
    result = spawnSync("npx", ["eslint", target, "--config", tmpConfig, ...eslintArgs], { cwd: context.directory, encoding: "utf-8", timeout: 60000 });
    try { fs.unlinkSync(tmpConfig); } catch { /* ignore */ }
  }
}

if (result.status !== 0) {
  const output = result.stdout || result.stderr || "";
  // Distinguish "no issues" from actual errors
  if (output.includes("0 errors") || output.includes("no issues")) {
    return `✓ No lint errors found.`;
  }
  return `Lint errors found:\n${output}`;
}
return `✓ No lint errors found.`;
```

## BUG #6 — e026: phase_complete Does Not Advance Phase

**File:** `src/tools/phase-complete.ts`, lines 20-22
**Error:** Phase stays at 'analyze' instead of advancing
**Severity:** HIGH

### Root Cause
Off-by-one error in PHASE_ORDER indexing. `phaseComplete` completes the CURRENT phase (passed as arg) but then advances by `idx + 1` from `PHASE_ORDER.indexOf(currentPhase)`. This means if you call `phase_complete phase=analyze`:
- `currentPhase = getCurrentPhase()` → returns `"idle"` (initial state, NOT the phase being completed)
- `idx = PHASE_ORDER.indexOf("idle")` → 0
- `nextPhase = PHASE_ORDER[1]` → `"analyze"`

So it never advances past `analyze`.

### Fix Required

```typescript
// BEFORE (line 11-26):
execute: async (args, context) => {
  const currentPhase = getCurrentPhase();
  const validTargets = PHASE_TRANSITIONS[currentPhase] || [];

  if (!validTargets.includes(args.phase)) {
    return `[BLOCKED] Cannot complete phase "${args.phase}" from "${currentPhase}".
Valid next phase: ${validTargets.join(", ") || "none"}`;
  }

  const idx = PHASE_ORDER.indexOf(currentPhase as typeof PHASE_ORDER[number]);
  const nextPhase: string = PHASE_ORDER[Math.min(idx + 1, PHASE_ORDER.length - 1)];
  setCurrentPhase(nextPhase);

  return `Phase "${args.phase}" complete.
Evidence: ${args.evidence}
Now in phase: ${nextPhase}`;
}

// AFTER:
execute: async (args, context) => {
  const currentPhase = getCurrentPhase();
  
  // args.phase is the phase the user wants to complete
  // Validate: args.phase must be the CURRENT phase
  if (args.phase !== currentPhase) {
    return `[BLOCKED] Cannot complete phase "${args.phase}". Current phase is "${currentPhase}".
You must complete phases in order.`;
  }

  const validTargets = PHASE_TRANSITIONS[currentPhase] || [];
  if (validTargets.length === 0) {
    return `Phase "${args.phase}" is already complete. No further phases.`;
  }

  // Find the NEXT phase (first valid transition target)
  const nextPhase = validTargets[0]; // e.g., for "idle" → ["analyze"], next is "analyze"
  setCurrentPhase(nextPhase);

  return `Phase "${args.phase}" complete.
Evidence: ${args.evidence}
Now in phase: ${nextPhase}`;
```

```typescript
    const dir = join(process.cwd(), ".hermes");
    if (!existsSync(dir)) mkdirSync(dir, { recursive: true });
    const phaseFile = join(dir, "phase.json");
    writeFileSync(phaseFile, JSON.stringify({ phase, updatedAt: new Date().toISOString() }));
  } catch { /* ignore */ }
}

export function getCurrentPhase(): string {
  if (!_phaseLoaded) {
    _currentPhase = loadPhaseFromDisk();
    _phaseLoaded = true;
  }
  return _currentPhase;
}

export function setCurrentPhase(phase: string): void {
  _currentPhase = phase;
  _phaseLoaded = true;
  savePhaseToDisk(phase);
}
```
